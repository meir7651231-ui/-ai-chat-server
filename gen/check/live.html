function cockpitDaysSince(iso, todayIso, T) {
  const MS_DAY = T.k1;
  if (!iso) return Infinity;
  const t = new Date(iso + 'T12:00:00').getTime();
  const now = new Date(todayIso + 'T12:00:00').getTime();

function countBy(items, key) {
    const m = new Map();
    for (const it of items) {
        const k = key(it);
        m.set(k, (m.get(k) ?? 0) + 1);
